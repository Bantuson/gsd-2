    Create packages/mission-control/src/server/workspace-api.ts:

    ```typescript
    import { homedir } from "node:os";
    import { join } from "node:path";
    import { mkdir } from "node:fs/promises";
    import { execFile } from "node:child_process";
    import { promisify } from "node:util";

    const execFileAsync = promisify(execFile);

    /** Override for testing */
    let _workspacePathOverride: string | null = null;
    export function _setWorkspacePathOverride(path: string | null): void {
      _workspacePathOverride = path;
    }

    export function getWorkspacePath(overridePath?: string): string {
      if (overridePath) return overridePath;
      if (_workspacePathOverride) return _workspacePathOverride;
      const home = homedir();
      return process.platform === "win32"
        ? join(process.env.USERPROFILE ?? home, "GSD Projects")
        : join(home, "GSD Projects");
    }

    export async function createProject(
      name: string,
      workspacePath: string
    ): Promise<{ projectPath: string }> {
      const projectPath = join(workspacePath, name);
      await mkdir(projectPath, { recursive: true });
      await execFileAsync("git", ["init"], { cwd: projectPath });
      return { projectPath };
    }
    ```
