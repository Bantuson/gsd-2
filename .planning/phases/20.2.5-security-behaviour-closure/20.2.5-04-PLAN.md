<objective>
Harden the two core security functions that underpin all path safety in Mission Control:
`validatePath` in fs-api.ts and `freePort` in kill-port.ts.

validatePath currently uses `resolve()` (does not follow symlinks) and `startsWith(root)` (no
trailing separator — sibling prefix bypass). Both flaws allow escape from the workspace root.

freePort currently kills any process on the port and sleeps, then returns the port number —
a classic TOCTOU race where another process can squat the port between kill and bind.

Purpose: Fix the foundation that all file-operation APIs depend on. Plans 05 and 06 build on top.
Output: Hardened validatePath and atomic freePort.
</objective>

```typescript
import { realpathSync } from "node:fs";
import path from "node:path";

export function validatePath(input: string, root: string): string {
  // Reject null bytes immediately — they cause undefined behaviour in OS path APIs
  if (input.includes("\x00")) {
    throw new Error("Invalid path");
  }

  // Resolve root to its real path (follow symlinks on the root itself)
  let resolvedRoot: string;
  try {
    resolvedRoot = realpathSync(root);
  } catch {
    throw new Error("Invalid path");
  }

  // Normalize the input relative to root (handles ../ sequences)
  const normalized = path.resolve(root, input);

  // Now resolve the normalized path to its real path (follows symlinks in input)
  let resolved: string;
  try {
    resolved = realpathSync(normalized);
  } catch {
    // If realpathSync fails, the path does not exist — still check the normalized form
    // against resolvedRoot to prevent traversal to non-existent-but-dangerous paths
    resolved = normalized;
  }

  // Critical: compare against resolvedRoot + path.sep to prevent sibling prefix bypass
  // e.g., /home/user/project-evil starts with /home/user/project but NOT /home/user/project/
  if (!resolved.startsWith(resolvedRoot + path.sep) && resolved !== resolvedRoot) {
    throw new Error("Invalid path");
  }

  return resolved;
}
```

```typescript
import { execFileAsync } from "./kill-port-exec"; // existing helper for kill-port command
import * as net from "node:net";

/**
 * Atomically finds and confirms a free port.
 *
 * Strategy:
 * 1. If the port is in use, kill the process using it (existing logic).
 * 2. Verify the port is free by briefly binding a probe socket.
 * 3. Return the port number.
 *
 * This does NOT hold the socket open — Bun.serve() will bind it properly.
 * The race window is minimized to microseconds (kill → verify → return).
 *
 * For port 0: bind a socket, capture the OS-assigned port, close the socket,
 * return that port. The caller must bind quickly to avoid squatting.
 */
export async function freePort(port: number): Promise<number> {
  if (port === 0) {
    return getEphemeralPort();
  }

  // Kill any process using this port (existing behaviour — keep this logic)
  await killProcessOnPort(port).catch(() => {
    // Ignore errors — port may already be free
  });

  // Verify the port is actually free by attempting to bind
  // Retry up to 3 times with 50ms backoff if still in use
  for (let attempt = 0; attempt < 3; attempt++) {
    const isFree = await checkPortFree(port);
    if (isFree) return port;
    await new Promise(r => setTimeout(r, 50));
  }

  // If still not free after 3 attempts, pick a different port atomically
  console.warn(`[kill-port] Port ${port} still in use after kill — using ephemeral port`);
  return getEphemeralPort();
}

/** Bind port 0 atomically, capture OS-assigned port, release socket. */
function getEphemeralPort(): Promise<number> {
  return new Promise((resolve, reject) => {
    const server = net.createServer();
    server.listen(0, "127.0.0.1", () => {
      const addr = server.address() as net.AddressInfo;
      const port = addr.port;
      server.close(() => resolve(port));
    });
    server.on("error", reject);
  });
}

/** Check if a port is free by trying to bind to it briefly. */
function checkPortFree(port: number): Promise<boolean> {
  return new Promise((resolve) => {
    const server = net.createServer();
    server.once("error", () => resolve(false));
    server.listen(port, "127.0.0.1", () => {
      server.close(() => resolve(true));
    });
  });
}
```

Keep the existing `killProcessOnPort` helper (or `killProcess` — whatever the current internal
