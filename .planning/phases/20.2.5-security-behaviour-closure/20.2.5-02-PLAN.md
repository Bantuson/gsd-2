**B51 — WebSocket upgrade requires token:**
```ts
// Attempt WS connection to a known WS port without auth token
// First register a window to get a WS port
const regRes = await makeRequest(server.baseUrl, "/api/window/register", {
  method: "POST",
  body: JSON.stringify({ windowId: "ws-auth-test" }),
});
const { wsPort } = await regRes.json();

// Now try to connect without the per-launch token
const ws = new WebSocket(`ws://127.0.0.1:${wsPort}`);
await new Promise<void>((resolve, reject) => {
  const timer = setTimeout(() => { ws.close(); resolve(); }, 2000);
  ws.onopen = () => {
    clearTimeout(timer);
    reject(new Error("WS should have been rejected without token"));
  };
  ws.onerror = () => { clearTimeout(timer); resolve(); };
  ws.onclose = (e) => {
    clearTimeout(timer);
    // Close code 1008 = policy violation = rejected
    expect(e.code).toBeOneOf([1008, 1002, 4001]);
    resolve();
  };
});
```
