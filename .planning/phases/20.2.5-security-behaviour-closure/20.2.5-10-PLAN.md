**server.ts — Rate limiter (B44):**
Add a simple in-memory rate limiter near the top of server.ts, after LAUNCH_TOKEN generation:
```typescript
// T-NET-02 B44: Basic rate limiter — 100 requests per second per IP
const rateLimitMap = new Map<string, { count: number; resetAt: number }>();
const RATE_LIMIT = 100; // requests per window
const RATE_WINDOW_MS = 1000; // 1 second

function checkRateLimit(ip: string): boolean {
  const now = Date.now();
  const entry = rateLimitMap.get(ip);
  if (!entry || now > entry.resetAt) {
    rateLimitMap.set(ip, { count: 1, resetAt: now + RATE_WINDOW_MS });
    return true; // within limit
  }
  entry.count++;
  if (entry.count > RATE_LIMIT) {
    return false; // over limit
  }
  return true;
}
```

In the `fetch` handler, add rate limit check after Host validation:
```typescript
// T-NET-02 B44: Rate limiting
const clientIp = req.headers.get("x-real-ip") ?? "127.0.0.1";
if (!checkRateLimit(clientIp)) {
  return new Response(JSON.stringify({ error: "Too many requests" }), {
    status: 429,
    headers: { "Content-Type": "application/json", "Retry-After": "1" },
  });
}
```
