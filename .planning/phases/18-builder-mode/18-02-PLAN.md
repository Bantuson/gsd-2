```typescript
// AppShell now has: useBuilderMode(), sendMessage from useSessionManager
// ADD: handleBuilderSend — async wrapper around sendMessage
// ADD: useState for routingBadgeState: { intent, originalMessage, sentAs } | null
// ADD: useState for phaseGateState: { originalMessage } | null

const handleBuilderSend = useCallback(async (message: string) => {
  if (!builderMode) { sendMessage(message); return; }
  // Show loading state on input (isClassifying state)
  setIsClassifying(true);
  try {
    const stateContent = JSON.stringify(state?.projectState ?? '');
    // Abort after 1500ms
    const controller = new AbortController();
    const timer = setTimeout(() => controller.abort(), 1500);
    const res = await fetch('/api/classify-intent', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ message, stateContext: stateContent }),
      signal: controller.signal,
    }).catch(() => null);
    clearTimeout(timer);

    const intent = res?.ok ? (await res.json().catch(() => null))?.intent : null;

    if (intent === 'UI_PHASE_GATE') {
      setPhaseGateState({ originalMessage: message });
      return;  // do NOT send the message
    }

    const sentAs = intent === 'GSD_COMMAND' ? '/gsd auto' : message;
    setRoutingBadgeState({ intent: intent ?? 'GENERAL_CODING', originalMessage: message, sentAs });
    sendMessage(sentAs);
  } finally {
    setIsClassifying(false);
  }
}, [builderMode, sendMessage, state]);
// Wire: pass onChatSend={handleBuilderSend} to SingleColumnView
// Also: pass routingBadgeState and phaseGateState down to SingleColumnView → ChatPanel
```

From auth.json format (written by Phase 16):
```json
{ "provider": "anthropic" | "openrouter" | "github-copilot" | "api-key", "access_token": "...", "api_key": "..." }
```
Skip classify when provider is "anthropic" (OAuth — cannot call api.anthropic.com) or "github-copilot".
Only classify when provider is "openrouter" or "api-key".

From INTENT_SYSTEM_PROMPT (from research):
```
You are a routing classifier for a developer tool.
Given the user message and current GSD project state, classify the intent as exactly one of:
- GSD_COMMAND (user wants to run a gsd workflow command like start, discuss, review, plan, execute)
- PHASE_QUESTION (user has a question about the current plan, decisions, or project state)
- GENERAL_CODING (user wants to write or modify code directly)
- UI_PHASE_GATE (user wants to build a visual frontend — CSS, component, page, layout, design — but no design context exists in the project state)

Respond with JSON only: { "intent": "<one of the four values>" }
```
